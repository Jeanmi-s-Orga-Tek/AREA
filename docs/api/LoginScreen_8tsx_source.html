8import React, { useState, useMemo, useEffect } from "react";
9import { login, fetchOAuthProviders, initiateOAuthLogin, OAuthProvider } from "../services/auth";
10import NotificationContainer, { NotificationItem } from "../components/NotificationContainer";
11import "./LoginScreen.css";

14type Star = {
15 id: string;
16 top: number;
17 left: number;
18 size: number;
19 delay: number;
20 dur: number;
21 alpha: number;
22};

24const LoginScreen: React.FC = () => {
25 const [email, setEmail] = useState("");
26 const [password, setPassword] = useState("");
27 const [notifications, setNotifications] = useState<NotificationItem[]>([]);
28 const [loading, setLoading] = useState(false);
29 const [oauthProviders, setOauthProviders] = useState<OAuthProvider[]>([]);
30 const [loadingProviders, setLoadingProviders] = useState(true);
31
32 const addNotification = (message: string, type: "success" | "error") => {
33 const id = `notif-${Date.now()}-${Math.random().toString(16).slice(2)}`;
34 setNotifications((prev) => [...prev, { id, message, type }]);
35 };
36
37 const removeNotification = (id: string) => {
38 setNotifications((prev) => prev.filter((n) => n.id !== id));
39 };
40
41 useEffect(() => {
42 const loadProviders = async () => {
43 try {
44 const providers = await fetchOAuthProviders();
45 setOauthProviders(providers.filter(p => p.available && p.flows.web));
46 } catch (err) {
47 console.error("Failed to load OAuth providers:", err);
48 } finally {
49 setLoadingProviders(false);
50 }
51 };
52
53 loadProviders();
54 }, []);
55
56 const stars = useMemo<Star[]>(() => {
57 const count = 120;
58 const rnd = (min: number, max: number) => min + Math.random() * (max - min);
59
60 return Array.from({ length: count }, (_, i) => ({
61 id: `s-${i}-${Math.random().toString(16).slice(2)}`,
62 top: rnd(0, 100),
63 left: rnd(0, 100),
64 size: rnd(1, 2.5),
65 delay: rnd(0, 6),
66 dur: rnd(1.2, 4.5),
67 alpha: rnd(0.25, 0.9),
68 }));
69 }, []);
70
71 const handleSubmit = async (e: React.FormEvent) => {
72 e.preventDefault();
73
74 if (!email || !password) {
75 addNotification("Tous les champs sont requis", "error");
76 return;
77 }
78
79 setLoading(true);
80
81 try {
82 await login({ email, password });
83 addNotification("Connexion réussie !", "success");
84 setTimeout(() => window.location.href = "/", 1000);
85 } catch (err) {
86 addNotification(err instanceof Error ? err.message : "Une erreur est survenue", "error");
87 } finally {
88 setLoading(false);
89 }
90 };
91
92 const handleOAuthLogin = async (providerId: string) => {
93 setLoading(true);
94
95 try {
96 await initiateOAuthLogin(providerId);
97 } catch (err) {
98 addNotification(err instanceof Error ? err.message : `Erreur lors de la connexion OAuth`, "error");
99 setLoading(false);
100 }
101 };
102
103 const renderProviderIcon = (provider: OAuthProvider) => {
104 const icon = provider.icon;
105
106 if (/^https?:\/\//i.test(icon)) {
107 return (
108 <img
109 src={icon}
110 alt={`${provider.name} logo`}
111 style={{
112 width: "24px",
113 height: "24px",
114 objectFit: "contain",
115 display: "inline-block",
116 marginRight: "8px",
117 }}
118 />
119 );
120 }
121
122 if (/^\s*<svg[\s>]/i.test(icon)) {
123 return (
124 <span
125 aria-label={`${provider.name} logo`}
126 role="img"
127 style={{
128 width: "24px",
129 height: "24px",
130 display: "inline-block",
131 marginRight: "8px",
132 }}
133 dangerouslySetInnerHTML={{ __html: icon }}
134 />
135 );
136 }
137
138 return <span style={{ marginRight: "8px" }}>{icon}</span>;
139 };
140
141 return (
142 <div className="login-container">
143 <NotificationContainer
144 notifications={notifications}
145 onRemove={removeNotification}
146 />
147 <div className="login-background" aria-hidden="true">
148 {stars.map((s) => (
149 <span
150 key={s.id}
151 className="star"
152 style={{
153 top: `${s.top}%`,
154 left: `${s.left}%`,
155 width: `${s.size}px`,
156 height: `${s.size}px`,
157 opacity: s.alpha,
158 animationDelay: `${s.delay}s`,
159 animationDuration: `${s.dur}s`,
160 }}
161 />
162 ))}
163 </div>
164 <div className="login-form-card">
165 <h1 className="login-title">Connexion</h1>
166 <p className="login-subtitle">Connectez-vous à votre compte AREA</p>
167
168
169 <form onSubmit={handleSubmit} className="login-form">
170 <div className="login-form-group">
171 <label htmlFor="email" className="login-label">
172 Email
173 </label>
174 <input
175 type="email"
176 id="email"
177 value={email}
178 onChange={(e) => setEmail(e.target.value)}
179 className="login-input"
180 placeholder="votre@email.com"
181 disabled={loading}
182 autoComplete="email"
183 />
184 </div>
185
186 <div className="login-form-group">
187 <label htmlFor="password" className="login-label">
188 Mot de passe
189 </label>
190 <input
191 type="password"
192 id="password"
193 value={password}
194 onChange={(e) => setPassword(e.target.value)}
195 className="login-input"
196 placeholder="••••••••"
197 disabled={loading}
198 autoComplete="current-password"
199 />
200 </div>
201
202 <button
203 type="submit"
204 className="login-button"
205 disabled={loading}
206 >
207
208 {loading ? "Connexion en cours..." : "Se connecter"}
209 </button>
210 </form>
211
212 <div className="login-divider">
213 <span>OU</span>
214 </div>
215
216 <div className="login-oauth-buttons">
217 {loadingProviders ? (
218 <p style={{ textAlign: "center", color: "#999" }}>
219 Chargement des providers...
220 </p>
221 ) : oauthProviders.length === 0 ? (
222 <p style={{ textAlign: "center", color: "#999" }}>
223 Aucun provider OAuth2 disponible
224 </p>
225 ) : (
226 oauthProviders.map((provider) => (
227 <button
228 key={provider.id}
229 onClick={() => handleOAuthLogin(provider.id)}
230 className="login-oauth-button"
231 disabled={loading}
232 style={{
233 borderLeft: `4px solid ${provider.color}`,
234 }}
235 >
236 {renderProviderIcon(provider)}
237 Se connecter avec {provider.name}
238 </button>
239 ))
240 )}
241 </div>
242
243 <div className="login-footer">
244 <p className="login-footer-text">
245 Vous n'avez pas de compte ?{" "}
246 <a href="/register" className="login-link">
247 S'inscrire
248 </a>
249 </p>
250 </div>
251 </div>
252 </div>
253 );
254};
255
256export default LoginScreen;
