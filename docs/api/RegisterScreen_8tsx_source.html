13type Star = {
14 id: string;
15 top: number;
16 left: number;
17 size: number;
18 delay: number;
19 dur: number;

23const RegisterScreen: React.FC = () => {
24 const [email, setEmail] = useState("");
25 const [username, setUsername] = useState("");
26 const [password, setPassword] = useState("");
27 const [confirmPassword, setConfirmPassword] = useState("");
28 const [notifications, setNotifications] = useState<NotificationItem[]>([]);
29 const [loading, setLoading] = useState(false);
30 const [oauthProviders, setOauthProviders] = useState<OAuthProvider[]>([]);
31 const [loadingProviders, setLoadingProviders] = useState(true);
32
33 const addNotification = (message: string, type: "success" | "error") => {
34 const id = `notif-${Date.now()}-${Math.random().toString(16).slice(2)}`;
35 setNotifications((prev) => [...prev, { id, message, type }]);
36 };
37
38 const removeNotification = (id: string) => {
39 setNotifications((prev) => prev.filter((n) => n.id !== id));
40 };
41
42 const stars = useMemo<Star[]>(() => {
43 const count = 120;
44 const rnd = (min: number, max: number) => min + Math.random() * (max - min);
45
46 return Array.from({ length: count }, (_, i) => ({
47 id: `s-${i}-${Math.random().toString(16).slice(2)}`,
48 top: rnd(0, 100),
49 left: rnd(0, 100),
50 size: rnd(1, 2.5),
51 delay: rnd(0, 6),
52 dur: rnd(1.2, 4.5),
53 alpha: rnd(0.25, 0.9),
54 }));
55 }, []);
56
57 useEffect(() => {
58 const loadProviders = async () => {
59 try {
60 const providers = await fetchOAuthProviders();
61 setOauthProviders(providers.filter(p => p.available && p.flows.web));
62 } catch (err) {
63 console.error("Failed to load OAuth providers:", err);
64 } finally {
65 setLoadingProviders(false);
66 }
67 };
68
69 loadProviders();
70 }, []);
71
72
73 const handleSubmit = async (e: React.FormEvent) => {
74 e.preventDefault();
75
76 if (!email || !username || !password || !confirmPassword) {
77 addNotification("Tous les champs sont requis", "error");
78 return;
79 }
80
81 if (password !== confirmPassword) {
82 addNotification("Les mots de passe ne correspondent pas", "error");
83 return;
84 }
85
86 if (password.length < 6) {
87 addNotification("Le mot de passe doit contenir au moins 6 caractères", "error");
88 return;
89 }
90
91 setLoading(true);
92
93 try {
94 await register({ email, name: username, new_password: password });
95 addNotification("Inscription réussie ! Redirection vers la page de connexion...", "success");
96 setEmail("");
97 setUsername("");
98 setPassword("");
99 setConfirmPassword("");
100 setTimeout(() => {
101 window.location.href = "/login";
102 }, 2000);
103 } catch (err) {
104 addNotification(err instanceof Error ? err.message : "Une erreur est survenue", "error");
105 } finally {
106 setLoading(false);
107 }
108 };
109
110 const handleOAuthLogin = async (providerId: string) => {
111 setLoading(true);
112
113 try {
114 await initiateOAuthLogin(providerId);
115 } catch (err) {
116 addNotification(err instanceof Error ? err.message : `Erreur lors de l'inscription OAuth`, "error");
117 setLoading(false);
118 }
119 };
120
121 const renderProviderIcon = (provider: OAuthProvider) => {
122 const icon = provider.icon;
123
124 if (/^https?:\/\//i.test(icon)) {
125 return (
126 <img
127 src={icon}
128 alt={`${provider.name} logo`}
129 style={{
130 width: "24px",
131 height: "24px",
132 objectFit: "contain",
133 display: "inline-block",
134 marginRight: "8px",
135 }}
136 />
137 );
138 }
139
140 if (/^\s*<svg[\s>]/i.test(icon)) {
141 return (
142 <span
143 aria-label={`${provider.name} logo`}
144 role="img"
145 style={{
146 width: "24px",
147 height: "24px",
148 display: "inline-block",
149 marginRight: "8px",
150 }}
151 dangerouslySetInnerHTML={{ __html: icon }}
152 />
153 );
154 }
155
156 return <span style={{ marginRight: "8px" }}>{icon}</span>;
157 };
158
159 return (
160 <div className="register-container">
161 <NotificationContainer
162 notifications={notifications}
163 onRemove={removeNotification}
164 />
165 <div className="register-background" aria-hidden="true">
166 {stars.map((s) => (
167 <span
168 key={s.id}
169 className="star"
170 style={{
171 top: `${s.top}%`,
172 left: `${s.left}%`,
173 width: `${s.size}px`,
174 height: `${s.size}px`,
175 opacity: s.alpha,
176 animationDelay: `${s.delay}s`,
177 animationDuration: `${s.dur}s`,
178 }}
179 />
180 ))}
181 </div>
182 <div className="register-form-card">
183 <h1 className="register-title">Inscription</h1>
184 <p className="register-subtitle">Créez votre compte AREA</p>
185
186
187 <form onSubmit={handleSubmit} className="register-form">
188 <div className="register-form-group">
189 <label htmlFor="email" className="register-label">
190 Email
191 </label>
192 <input
193 type="email"
194 id="email"
195 value={email}
196 onChange={(e) => setEmail(e.target.value)}
197 className="register-input"
198 placeholder="votre@email.com"
199 disabled={loading}
200 />
201 </div>
202
203 <div className="register-form-group">
204 <label htmlFor="username" className="register-label">
205 Nom d'utilisateur
206 </label>
207 <input
208 type="text"
209 id="username"
210 value={username}
211 onChange={(e) => setUsername(e.target.value)}
212 className="register-input"
213 placeholder="John Doe"
214 disabled={loading}
215 />
216 </div>
217
218 <div className="register-form-group">
219 <label htmlFor="password" className="register-label">
220 Mot de passe

222 <input
223 type="password"
224 id="password"

229 disabled={loading}

231 </div>
232
233 <div className="register-form-group">
234 <label htmlFor="confirmPassword" className="register-label">

237 <input

239 id="confirmPassword"

241 onChange={(e) => setConfirmPassword(e.target.value)}
242 className="register-input"
243 placeholder="••••••••"
244 disabled={loading}
245 />
246 </div>
247
248 <button
249 type="submit"
250 className="register-button"

253 {loading ? "Inscription en cours..." : "S'inscrire"}
254 </button>
255 </form>
256
257 <div className="register-divider">
258 <span>OU</span>
259 </div>
260
261 <div className="register-oauth-buttons">
262 {loadingProviders ? (
263 <p style={{ textAlign: "center", color: "#999" }}>Chargement des providers...</p>
264 ) : oauthProviders.length === 0 ? (
265 <p style={{ textAlign: "center", color: "#999" }}>Aucun provider OAuth2 disponible</p>

267 oauthProviders.map((provider) => (
268 <button
269 key={provider.id}

272 disabled={loading}

274 borderLeft: `4px solid ${provider.color}`,
275 }}
276 >
277 {renderProviderIcon(provider)}
278 S'inscrire avec {provider.name}
279 </button>
280 ))
281 )}
282 </div>
283
284 <div className="register-footer">
285 <p className="register-footer-text">

288 Se connecter

290 </p>
291 </div>
292 </div>
293 </div>
294 );
295};
