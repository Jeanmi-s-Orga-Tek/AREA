8import React, { useEffect, useState } from "react";

13const OAuthCallbackScreen: React.FC = () => {
14 const [error, setError] = useState<string | null>(null);
15 const [loading, setLoading] = useState(true);
16 const [stars, setStars] = useState<Array<{ id: number; style: React.CSSProperties }>>([]);
17
18 useEffect(() => {
19 const generateStars = () => {
20 const newStars = [];
21 for (let i = 0; i < 100; i++) {
22 const size = Math.random() * 3 + 1;
23 newStars.push({
24 id: i,
25 style: {
26 width: `${size}px`,
27 height: `${size}px`,
28 left: `${Math.random() * 100}%`,
29 top: `${Math.random() * 100}%`,
30 animationDelay: `${Math.random() * 5}s`,
31 animationDuration: `${Math.random() * 3 + 2}s`,
32 },
33 });
34 }
35 setStars(newStars);
36 };
37
38 generateStars();
39 }, []);
40
41 useEffect(() => {
42 const processOAuthCallback = async () => {
43 try {
44 if (window.location.hostname === "127.0.0.1") {
45 const newUrl = window.location.href.replace("127.0.0.1", "localhost");
46 window.location.href = newUrl;
47 return;
48 }
49
50 const urlParams = new URLSearchParams(window.location.search);
51
52 const hash = window.location.hash;
53 const hashContent = hash.replace(/^#/, '');
54
55 let hashToken = null;
56 let hashState = null;
57 if (hashContent) {
58 const hashPairs = hashContent.split('&');
59 for (const pair of hashPairs) {
60 const [key, value] = pair.split('=');
61 if (key === 'token') hashToken = decodeURIComponent(value);
62 if (key === 'state') hashState = decodeURIComponent(value);
63 }
64 }
65
66 let code = urlParams.get("code") || urlParams.get("token") || hashToken;
67 let state = urlParams.get("state") || hashState;
68 const errorParam = urlParams.get("error");
69
70 if (errorParam) {
71 const errorDescription = urlParams.get("error_description") || errorParam;
72 throw new Error(`Erreur OAuth: ${errorDescription}`);
73 }
74
75 if (!code) {
76 throw new Error("Code manquant dans l'URL de callback");
77 }
78
79 const stateValue = state || "";
80
81 const pathParts = window.location.pathname.split("/");
82 const provider = pathParts[pathParts.length - 1];
83
84 if (!provider) {
85 throw new Error("Provider non identifié dans l'URL");
86 }
87
88 const connectingService = sessionStorage.getItem("connecting_service");
89 const connectingProvider = sessionStorage.getItem("connecting_service_provider");
90
91 console.log("=== Vérification connexion service ===");
92 console.log("connecting_service:", connectingService);
93 console.log("connecting_service_provider:", connectingProvider);
94 console.log("current provider:", provider);
95
96 if (connectingService && connectingProvider === provider) {
97 console.log(" Mode CONNEXION SERVICE (pas login)");
98 console.log("Service à connecter:", connectingService);
99 try {
100 console.log("Appel API connectService avec code:", code.substring(0, 20) + "...");
101 await connectService(connectingService, code, "web");
102 console.log("Service connecté avec succès!");
103
104 sessionStorage.removeItem("connecting_service");
105 sessionStorage.removeItem("connecting_service_provider");
106 console.log(" SessionStorage nettoyé");
107
108 console.log("Redirection vers /services");
109 window.location.href = "/services";
110 return;
111 } catch (serviceErr) {
112 console.error("Erreur lors de la connexion du service:", serviceErr);
113 console.error("Details:", serviceErr);
114 throw serviceErr;
115 }
116 }
117
118 console.log("Mode LOGIN OAuth");
119 console.log("Authentification OAuth avec provider:", provider);
120 await handleOAuthCallback(provider, code, stateValue);
121 console.log(" Authentification réussie");
122
123 console.log("Redirection vers /");
124 window.location.href = "/";
125 } catch (err) {
126 console.error("OAuth callback error:", err);
127 setError(err instanceof Error ? err.message : "Une erreur est survenue");
128 setLoading(false);
129
130 setTimeout(() => {
131 window.location.href = "/login";
132 }, 3000);
133 }
134 };
135
136 processOAuthCallback();
137 }, []);
138
139 if (loading) {
140 return (
141 <div className="oauth-container">
142 <div className="oauth-background">
143 {stars.map((star) => (
144 <div key={star.id} className="star" style={star.style} />
145 ))}
146 </div>
147 <div className="oauth-card">
148 <div className="oauth-spinner"></div>
149 <h2 className="oauth-title">Authentification en cours...</h2>
150 <p className="oauth-message">Veuillez patienter pendant que nous vous connectons.</p>
151 </div>
152 </div>
153 );
154 }
155
156 if (error) {
157 return (
158 <div className="oauth-container">
159 <div className="oauth-background">
160 {stars.map((star) => (
161 <div key={star.id} className="star" style={star.style} />
162 ))}
163 </div>
164 <div className="oauth-card">
165 <h2 className="oauth-title">Erreur d'authentification</h2>
166 <p className="oauth-error-message">{error}</p>
167 <p className="oauth-message">Vous allez être redirigé vers la page de connexion...</p>
168 </div>
169 </div>
170 );
171 }
172
173 return null;
174};
